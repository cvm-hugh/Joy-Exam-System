import fs from 'node:fs/promises';
import { FileBlob, SpreadsheetFile } from '@oai/artifact-tool';

const inputPath = '/Users/contiference/Desktop/学业水平统考-英语成绩.xlsx';
const outDir = '/private/tmp/english_score_inspect';
await fs.mkdir(outDir, { recursive: true });
const input = await FileBlob.load(inputPath);
const wb = await SpreadsheetFile.importXlsx(input);

const summary = await wb.inspect({
  kind: 'workbook,sheet,table',
  maxChars: 12000,
  tableMaxRows: 5,
  tableMaxCols: 35,
  tableMaxCellChars: 80,
});
console.log('SUMMARY\n' + summary.ndjson);

const grades = wb.worksheets.getItem('成绩');
const used = grades.getUsedRange(true);
console.log('USED_RANGE', used.address ?? 'unknown');
const region = await wb.inspect({
  kind: 'table',
  sheetId: '成绩',
  range: 'A1:AB20',
  include: 'values,formulas',
  tableMaxRows: 20,
  tableMaxCols: 28,
  maxChars: 30000,
});
console.log('GRADES_HEAD\n' + region.ndjson);

const preview = await wb.render({ sheetName: '成绩', range: 'A1:U25', scale: 1, format: 'png' });
await fs.writeFile(`${outDir}/grades_preview.png`, new Uint8Array(await preview.arrayBuffer()));
console.log('PREVIEW', `${outDir}/grades_preview.png`);

const values = grades.getRange('A1:AB300').values;
const headers = values[0];
const rows = [];
const itemIdx = Array.from({ length: 11 }, (_, i) => 6 + i);
for (let i = 1; i < values.length; i++) {
  const row = values[i];
  const code = row[0];
  if (code === null || code === undefined || String(code).trim() === '') continue;
  const writing = row[17];
  const total = row[20];
  const writingZero = typeof writing === 'number' && writing === 0;
  const zeroItems = itemIdx.filter(j => typeof row[j] === 'number' && row[j] === 0).map(j => headers[j]);
  const lowTotal = typeof total === 'number' && total < 40;
  if (writingZero || zeroItems.length > 0 || lowTotal) {
    const reasons = [];
    if (writingZero) reasons.push('作文为0');
    if (zeroItems.length > 0) reasons.push(`其他单项为0：${zeroItems.join('、')}`);
    if (lowTotal) reasons.push('总分<40');
    rows.push({ sourceRow: i + 1, code, name: row[1], campus: row[3], ref: row[4], writing, zeroItems, total, reasons });
  }
}
console.log('CANDIDATE_COUNT', rows.length);
console.log(JSON.stringify(rows, null, 2));
