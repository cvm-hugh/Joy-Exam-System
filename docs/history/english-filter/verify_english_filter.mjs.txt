import { FileBlob, SpreadsheetFile } from '@oai/artifact-tool';

const path = process.argv[2] || new URL('./outputs/english_filter_20260912/学业水平统考-英语成绩.xlsx', import.meta.url).pathname;
const wb = await SpreadsheetFile.importXlsx(await FileBlob.load(path));
const out = wb.worksheets.getItem('英语筛选名单');
const vals = out.getRange('A1:V100').values;
console.log('SHEETS', (await wb.inspect({ kind: 'sheet', include: 'name' })).ndjson);
console.log('OUT_HEAD', JSON.stringify(vals.slice(0, 4)));
console.log('OUT_LAST', JSON.stringify(vals.slice(-3)));
console.log('OUT_RANGE', out.getUsedRange(true).address);
console.log('A_FORMAT', JSON.stringify(out.getRange('A1:A4').format.numberFormat));
const err = await wb.inspect({ kind: 'match', searchTerm: '#REF!|#DIV/0!|#VALUE!|#NAME\\?|#N/A|#NUM!|#NULL!|#SPILL!|#CALC!', options: { useRegex: true, maxResults: 300 }, summary: 'saved output error scan' });
console.log('ERRORS', err.ndjson);
